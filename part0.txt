/* ---------- Scoring ---------- */
// Secondary risk signals: option index -> pole, so Risk Attitude is read from more than 3 questions
const RISK2 = { 2: { 0: "C" }, 3: { 0: "B", 2: "C" }, 4: { 0: "C", 1: "B" }, 7: { 0: "B", 3: "C" }, 8: { 0: "B", 2: "C", 3: "C" }, 9: { 0: "C", 1: "B", 3: "B" }, 10: { 0: "C", 2: "C" }, 14: { 0: "C", 1: "B", 3: "C" }, 15: { 2: "C" } };

function score(ans) {
  const s = { T: 0, L: 0, V: 0, G: 0, C: 0, B: 0, F: 0, D: 0 };
  ans.forEach((a, i) => {
    if (i === 17 || a === null) return;
    const list = Array.isArray(a) ? a : [a];
    // Breadth evidence for Spread: how many industries/markets were selected in Q10-Q13
    if (i >= 9 && i <= 12 && !(i === 10 && list.includes(3))) {
      if (list.length >= 3) s.D += 1;
      else if (list.length === 1) s.F += 1;
    }
    list.forEach((o) => {
      const p = Q[i][1][o][1];
      if (p) s[p] += i === 5 ? 2 : 1; // Q6 counts double; multi-select adds one point per ticked option
      const p2 = RISK2[i] && RISK2[i][o];
      if (p2) s[p2] += 1;
    });
  });
  const code = [];
  const axes = AXES.map((ax) => {
    const [p1, p2] = ax.p, a = s[p1], b = s[p2];
    // Add-one smoothing: a short quiz never claims a certain 0% or 100%
    const pct1 = Math.round(((a + 1) / (a + b + 2)) * 100);
    const win = a > b ? p1 : b > a ? p2 : Q[ax.tie][1][ans[ax.tie]][1] || p1;
    code.push(win);
    return { ...ax, pct: [pct1, 100 - pct1], win, n: a + b };
  });
  return { code: code.join("-"), axes, level: LEVELS[ans[17]] };
}

    `<span class="font-semibold text-slate-700">Note.</span> Self-assessment only. Not financial, investment or tax advice.`)}

  <div class="noprint mt-8 flex flex-wrap justify-center gap-3">
    <button data-act="print" class="${btn}">Save as PDF</button>
    <button data-act="restart" class="rounded-xl border border-slate-200 bg-white px-5 py-2.5 text-sm font-medium text-slate-700 shadow-sm hover:bg-slate-50">↺ Retake</button>
  </div>
</main>`;
}

function render(keepScroll) {
  const y = window.scrollY;
  S.anim = !keepScroll;
  if (!keepScroll) S.lastSel = null;
  root.innerHTML = { landing, quiz, lead, results }[S.stage]();
  window.scrollTo(0, keepScroll ? y : 0);
  requestAnimationFrame(() => requestAnimationFrame(() => {
    root.querySelectorAll("[data-w]").forEach((el) => (el.style.width = el.dataset.w + "%"));
    S.prev = PAGES.map((pg) => (S.ans.slice(pg.range[0], pg.range[1]).filter((v) => v !== null).length / 6) * 100);
  }));
}

root.addEventListener("click", (ev) => {
  const t = ev.target.closest("[data-q],[data-act]");
  if (!t) return;
  if (t.dataset.q !== undefined) {
    const qi = +t.dataset.q, was = S.ans[qi];
    const o = +t.dataset.o;
    if (MULTI.has(qi)) {
      const cur = S.ans[qi] || [], nx = cur.includes(o) ? cur.filter((x) => x !== o) : [...cur, o].sort();
      S.ans[qi] = nx.length ? nx : null; S.lastSel = qi + "-" + o; return render(true);
    }
    S.ans[qi] = o; S.lastSel = qi + "-" + o; render(true);
    if (was === null) {
      const n = S.ans.findIndex((v, i) => i > qi && i < range()[1] && v === null);
      if (n > -1) setTimeout(() => { const el = document.getElementById("q" + n); if (el) el.scrollIntoView({ behavior: "smooth", block: "center" }); }, 280);
    }
    return;
  }
  const act = t.dataset.act;
  if (act === "poke") { t.classList.remove("jump"); void t.offsetWidth; t.classList.add("jump"); return; }
  if (act === "start") { S.stage = "quiz"; S.page = 0; }
  else if (act === "next") {
    if (!pageDone()) {
      S.showErr = true; render(true);
      const m = S.ans.findIndex((v, i) => i >= range()[0] && i < range()[1] && v === null);
      const el = document.getElementById("q" + m); if (el) el.scrollIntoView({ behavior: "smooth", block: "center" });
      return;
    }
    S.showErr = false;
    if (S.page < 2) S.page++; else S.stage = "lead";
  } else if (act === "back") {
    S.showErr = false;
    if (S.page > 0) S.page--; else S.stage = "landing";
  } else if (act === "print") { return window.print();
  } else if (act === "restart") {
    S.ans = Array(18).fill(null); S.page = 0; S.save = null; S.err = {}; S.stage = "landing";
  }
  render();
});

const sync = (ev) => { if (ev.target.name in S.form) S.form[ev.target.name] = ev.target.value; };
root.addEventListener("input", sync);
root.addEventListener("change", (ev) => {
  sync(ev);
  if (ev.target.name === "category") {
    render(true);
    const o = document.querySelector("[name=categoryOther]");
    if (o) o.focus();
  }
});

root.addEventListener("submit", async (ev) => {
  ev.preventDefault();
  if (S.saving) return;
  const f = S.form, err = {};
  if (!f.name.trim()) err.name = "Enter your full name.";
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(f.email.trim())) err.email = "Enter a valid professional email.";
  if (f.linkedin && !/^https?:\/\/.+/i.test(f.linkedin.trim())) err.linkedin = "Enter a full URL starting with https://";
  if (!f.category) err.category = "Select an investor category.";
  if (f.category === "Other" && !f.categoryOther.trim()) err.categoryOther = "Please specify your investor category.";
  if (!f.ticket) err.ticket = "Select a deployment capacity.";
  S.err = err;
  if (Object.keys(err).length) return render(true);
  S.saving = true; render(true);
  try {
    await fbReady;
    if (!S.fb) throw new Error("Cloud storage unavailable");
    const r = score(S.ans), a = A.find((x) => x[0] === r.code), { db, auth, fs } = S.fb;
    await fs.addDoc(fs.collection(db, "artifacts", appId, "public", "data", "investor_assessments"), {
      name: f.name.trim(), email: f.email.trim(), linkedin: f.linkedin.trim(), category: f.category, categoryOther: f.category === "Other" ? f.categoryOther.trim() : "", ticket: f.ticket,
      resultCode: r.code, animal: a[2], experienceLevel: r.level,
      axisPercentiles: Object.fromEntries(r.axes.map((x) => [x.label, { [x.names[0]]: x.pct[0], [x.names[1]]: x.pct[1] }])),
      answers: S.ans.map((v) => (Array.isArray(v) ? v.join(",") : String(v))), uid: auth.currentUser ? auth.currentUser.uid : null, createdAt: fs.serverTimestamp(),
    });
    S.save = "saved";
  } catch (e) { console.error(e); S.save = "failed"; }
  S.saving = false; S.stage = "results"; render();
});

render();
